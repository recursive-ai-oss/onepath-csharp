cmake_minimum_required(VERSION 3.20)
project(OnePathNativeHarness CXX)

# Black-box drivers for Recast/Detour/DetourCrowd and RVO2. Their releases are built and installed by
# tools/refs/refs.sh; this project only compiles our driver sources against those installs.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(RECAST_PREFIX "" CACHE PATH "Install prefix of the Recast/Detour release")
set(RVO2_PREFIX "" CACHE PATH "Install prefix of the RVO2 release")

find_path(RECAST_INCLUDE Recast.h PATHS ${RECAST_PREFIX}/include PATH_SUFFIXES recastnavigation NO_DEFAULT_PATH)
find_library(RECAST_LIB NAMES Recast PATHS ${RECAST_PREFIX}/lib ${RECAST_PREFIX}/lib64 NO_DEFAULT_PATH)
find_library(DETOUR_LIB NAMES Detour PATHS ${RECAST_PREFIX}/lib ${RECAST_PREFIX}/lib64 NO_DEFAULT_PATH)
find_library(DETOURCROWD_LIB NAMES DetourCrowd PATHS ${RECAST_PREFIX}/lib ${RECAST_PREFIX}/lib64 NO_DEFAULT_PATH)
find_library(DETOURTILECACHE_LIB NAMES DetourTileCache PATHS ${RECAST_PREFIX}/lib ${RECAST_PREFIX}/lib64 NO_DEFAULT_PATH)

if(RECAST_INCLUDE AND RECAST_LIB AND DETOUR_LIB AND DETOURCROWD_LIB AND DETOURTILECACHE_LIB)
  add_executable(recast-harness recast_harness.cpp)
  target_include_directories(recast-harness PRIVATE ${RECAST_INCLUDE})
  target_link_libraries(recast-harness PRIVATE ${DETOURTILECACHE_LIB} ${DETOURCROWD_LIB} ${DETOUR_LIB} ${RECAST_LIB})
  install(TARGETS recast-harness RUNTIME DESTINATION bin)
else()
  message(WARNING "Recast install not found under '${RECAST_PREFIX}': recast-harness skipped")
endif()

find_path(RVO_INCLUDE RVO.h PATHS ${RVO2_PREFIX}/include PATH_SUFFIXES RVO rvo2 NO_DEFAULT_PATH)
find_library(RVO_LIB NAMES RVO PATHS ${RVO2_PREFIX}/lib ${RVO2_PREFIX}/lib64 NO_DEFAULT_PATH)
if(RVO_INCLUDE AND RVO_LIB)
  find_package(OpenMP QUIET)
  add_executable(rvo-harness rvo_harness.cpp)
  target_include_directories(rvo-harness PRIVATE ${RVO_INCLUDE})
  target_link_libraries(rvo-harness PRIVATE ${RVO_LIB})
  if(OpenMP_CXX_FOUND)
    target_link_libraries(rvo-harness PRIVATE OpenMP::OpenMP_CXX)
  endif()
  install(TARGETS rvo-harness RUNTIME DESTINATION bin)
else()
  message(WARNING "RVO2 install not found under '${RVO2_PREFIX}': rvo-harness skipped")
endif()
